/**
 * SMART WAREHOUSE V2.0 - CPO DOWNTIME API
 * Script ini digunakan untuk membaca data dari sheet "PENERIMAAN(MANUAL)".
 * 
 * CARA MENGGUNAKAN:
 * 1. Buka file Google Sheets "CPO STOCK" (atau tempat sheet PENERIMAAN(MANUAL) berada).
 * 2. Klik Extensions (Ekstensi) > Apps Script.
 * 3. Copy-Paste seluruh kode ini ke dalam Code.gs.
 * 4. Klik Deploy > New Deployment.
 * 5. Pilih Web App, Execute as: Me, Who has access: Anyone.
 * 6. Copy URL Web App yang dihasilkan, lalu masukkan ke file `js/app-config.js` di baris `CPO_DOWNTIME_API_URL`.
 */

const SHEET_NAME = "PENERIMAAN(MANUAL)";
const TANK_SHEET_NAME = "DATA FISIK(AUTO)";
const INVENTORY_SHEET_NAME = "CPO DAILY(AUTO)";

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // =============== ROUTE: GET TANK PHYSICAL DATA ===============
    if (e && e.parameter && e.parameter.action === 'getTankData') {
      const sheet = ss.getSheetByName(TANK_SHEET_NAME);
      if (!sheet) {
        return ContentService.createTextOutput(JSON.stringify({
          success: false,
          message: "Sheet " + TANK_SHEET_NAME + " tidak ditemukan!"
        })).setMimeType(ContentService.MimeType.JSON);
      }

      // Read B3 to O9
      // Row 3 to 9 = 7 rows. Col 2 (B) to 15 (O) = 14 cols.
      const range = sheet.getRange(3, 2, 7, 14);
      const values = range.getValues();
      const tankResults = [];

      for (let i = 0; i < values.length; i++) {
        const row = values[i];
        let tName = String(row[0] || "").trim(); // B (index 0)
        let tStock = Number(row[3]) || 0;       // E (index 3)
        let tCap = Number(row[7]) || 0;         // I (index 7)
        let tAge = Number(row[10]) || 0;        // L (index 10)
        let tRemark = String(row[13] || "").trim(); // O (index 13)

        if (tName !== "") {
          tankResults.push({
            name: tName,
            stock: tStock,
            cap: tCap,
            age: tAge,
            remark: tRemark
          });
        }
      }

      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        data: tankResults
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // =============== ROUTE: GET INVENTORY DATA ===============
    if (e && e.parameter && e.parameter.action === 'getInventoryData') {
      const sheet = ss.getSheetByName(INVENTORY_SHEET_NAME);
      if (!sheet) {
        return ContentService.createTextOutput(JSON.stringify({
          success: false,
          message: "Sheet " + INVENTORY_SHEET_NAME + " tidak ditemukan!"
        })).setMimeType(ContentService.MimeType.JSON);
      }

      const startRow = 8;
      const endRow = sheet.getLastRow();
      
      if (endRow < startRow) {
        return ContentService.createTextOutput(JSON.stringify({
          success: true,
          data: []
        })).setMimeType(ContentService.MimeType.JSON);
      }

      // Read B8 to BN(lastRow)
      // B starts at 2, BN is 66 (difference is 65 columns)
      const numCols = 65; 
      const range = sheet.getRange(startRow, 2, endRow - startRow + 1, numCols);
      const values = range.getValues();
      const results = [];

      const formatDate = (val) => {
          if (!val) return "";
          // Check if it's a date object (even across contexts)
          if (Object.prototype.toString.call(val) === '[object Date]' || (val && typeof val.getTime === 'function')) {
              let d = val.getDate();
              let m = val.getMonth() + 1;
              let y = val.getFullYear();
              return (d < 10 ? '0' + d : d) + '-' + (m < 10 ? '0' + m : m) + '-' + y;
          }
          // Try to parse if it's a string looking like a date
          let strVal = String(val).trim();
          let parsed = new Date(strVal);
          if (!isNaN(parsed) && strVal.length > 10) {
              let d = parsed.getDate();
              let m = parsed.getMonth() + 1;
              let y = parsed.getFullYear();
              return (d < 10 ? '0' + d : d) + '-' + (m < 10 ? '0' + m : m) + '-' + y;
          }
          return strVal;
      };

      for (let i = 0; i < values.length; i++) {
        const row = values[i];
        const tanggal = formatDate(row[0]); // B (index 0)
        
        if (!tanggal) continue;

        // Data Stock TK1-TK7 : C (index 1) to I (index 7)
        // Penerimaan TK1-TK7 : J (index 8) to P (index 14)
        // Usage TK1-TK7 : Q (index 15) to W (index 21)
        // Cek Aktual TK1-TK7 : Y (index 23) to AE (index 29)
        // SAP TK1-TK7 : BH (index 58) to BN (index 64)

        const getArr = (startIdx, length) => {
            let arr = [];
            for(let j=0; j<length; j++) {
                arr.push(Number(row[startIdx + j]) || 0);
            }
            return arr;
        };

        results.push({
            tanggal: tanggal,
            data_stock: getArr(1, 7),
            penerimaan: getArr(8, 7),
            usage: getArr(15, 7),
            cek_aktual: getArr(23, 7),
            sap: getArr(58, 7)
        });
      }

      const resultObj = { success: true, data: results };
      if (e.parameter.callback) {
        return ContentService.createTextOutput(e.parameter.callback + "(" + JSON.stringify(resultObj) + ")")
          .setMimeType(ContentService.MimeType.JAVASCRIPT);
      }
      return ContentService.createTextOutput(JSON.stringify(resultObj))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // =============== ROUTE: DEFAULT (DOWNTIME DATA) ===============
    const sheet = ss.getSheetByName(SHEET_NAME);
    
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        message: "Sheet PENERIMAAN(MANUAL) tidak ditemukan!"
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // Data dimulai dari baris ke-4
    // Kolom:
    // B (2): TANGGAL
    // C (3): NOPOL
    // D (4): AWAL
    // E (5): AKHIR
    // F (6): DURASI E-D
    // G (7): NETTO
    // H (8): SUHU
    // I (9): TANKI
    // J (10): SHIFT
    // K (11): KETERANGAN
    
    const startRow = 4;
    const endRow = sheet.getLastRow();
    
    // Jika tidak ada data
    if (endRow < startRow) {
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        data: []
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // Ambil data dari B4 sampai K(LastRow)
    const range = sheet.getRange(startRow, 2, endRow - startRow + 1, 10);
    const values = range.getValues();
    
    const results = [];
    
    for (let i = 0; i < values.length; i++) {
      const row = values[i];
      
      let tanggal = row[0]; // B
      if (!tanggal) continue; // Skip jika tanggal kosong
      
      // Format tanggal ke string YYYY-MM-DD jika berbentuk object Date
      if (tanggal instanceof Date) {
        tanggal = Utilities.formatDate(tanggal, Session.getScriptTimeZone(), "yyyy-MM-dd");
      } else {
        tanggal = String(tanggal).trim();
      }
      
      let nopol = String(row[1] || "").trim(); // C
      let awal = row[2]; // D (Waktu/Jam)
      let akhir = row[3]; // E (Waktu/Jam)
      let durasi = row[4]; // F (Bisa string HH:mm:ss atau number)
      let netto = Number(row[5]) || 0; // G (NETTO)
      let suhu = Number(row[6]) || 0; // H (SUHU)
      let tanki = String(row[7] || "").trim(); // I (TANKI)
      let shift = String(row[8] || "").trim(); // J (SHIFT)
      let keterangan = String(row[9] || "").trim(); // K (KETERANGAN)
      
      // Proses konversi waktu ke string HH:mm untuk AWAL dan AKHIR
      const formatJam = (val) => {
        if (!val && val !== 0) return null;
        if (val instanceof Date) {
          let h = val.getHours();
          let m = val.getMinutes();
          return (h < 10 ? "0" + h : String(h)) + ":" + (m < 10 ? "0" + m : String(m));
        }
        
        // Penanganan format string atau teks
        let s = String(val).trim();
        if (s.includes(':')) {
           // Jika ada detik HH:MM:SS, buang detiknya
           let parts = s.split(':');
           return (parts[0].length === 1 ? "0" + parts[0] : parts[0]) + ":" + parts[1];
        }

        // Penanganan format decimal dari GSheets murni (misal 0.5 = 12:00)
        // Hindari angka negatif (seperti value kosong yang jadi tahun 1899)
        if (!isNaN(val) && val >= 0 && val < 1) {
            let totalMins = Math.round(val * 24 * 60);
            let h = Math.floor(totalMins / 60);
            let m = totalMins % 60;
            return (h < 10 ? "0" + h : String(h)) + ":" + (m < 10 ? "0" + m : String(m));
        }

        return null;
      };
      
      // Jika durasi dari sheet kosong, kita bisa hitung sendiri dari E - D
      let durasiMenit = 0;
      if (durasi && !isNaN(durasi) && durasi < 1) {
          durasiMenit = Math.round(durasi * 24 * 60);
      } else if (durasi && String(durasi).includes(':')) {
           let parts = String(durasi).split(':');
           durasiMenit = (parseInt(parts[0]) * 60) + parseInt(parts[1]);
      } else if (!isNaN(durasi)) {
          durasiMenit = Number(durasi);
      }

      // Jika masih 0 dan Awal Akhir ada
      if (durasiMenit === 0 && formatJam(awal) && formatJam(akhir)) {
          let s = formatJam(awal).split(':');
          let e = formatJam(akhir).split(':');
          let mS = (parseInt(s[0]) * 60) + parseInt(s[1]);
          let mE = (parseInt(e[0]) * 60) + parseInt(e[1]);
          
          let diff = mE - mS;
          if (diff < 0) diff += 1440; // Jika lewat tengah malam
          durasiMenit = diff;
      }

      results.push({
        tanggal: tanggal,
        nopol: nopol,
        start_time: formatJam(awal),
        end_time: formatJam(akhir),
        durasi_minutes: durasiMenit,
        netto_kg: netto,
        suhu: suhu,
        tanki: tanki,
        shift: shift,
        keterangan: keterangan
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      data: results
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
